CREATE DATABASE Digicoin;
USE Digicoin;

SET GLOBAL event_scheduler = ON;

DELIMITER $$

CREATE PROCEDURE desativar_campanhas_expiradas()
BEGIN
    -- Desativa produtos de campanhas expiradas
    UPDATE api_produto
    SET is_active = FALSE
    WHERE idCampanha_id IN (
        SELECT id FROM api_campanha
        WHERE dataFim < CURDATE() AND is_active = TRUE
    );

    -- Desativa desafios de campanhas expiradas
    UPDATE api_desafio
    SET is_active = FALSE
    WHERE idCampanha_id IN (
        SELECT id FROM api_campanha
        WHERE dataFim < CURDATE() AND is_active = TRUE
    );

    -- Desativa as próprias campanhas
    UPDATE api_campanha
    SET is_active = FALSE
    WHERE dataFim < CURDATE() AND is_active = TRUE;
END$$


DELIMITER ;

CREATE EVENT desativar_campanhas_event
ON SCHEDULE EVERY 1 DAY
STARTS NOW() -- Opcional: inicia o evento imediatamente
DO
    CALL desativar_campanhas_expiradas();
